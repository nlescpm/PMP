\section{Project lifecycle checklist}
\label{app:checklist}

\paragraph{Project Initiation}
\begin{itemize}
    \item[$\square$] Section Heads: Assign project to a SH 
    \item[$\square$] Finance: Register project in AFAS
    \item[$\square$] SH: Assign Lead RSE and eScience project team
    \item[$\square$] Lead RSE: Create RSD page
    \item[$\square$] SH: Hold administrative kick-off meeting
    \item[$\square$] Lead RSE: Organize project kick-off meeting
    \item[$\square$] SH: Arrange legal agreements
    \item[$\square$] Lead RSE: Draft technology plan
\end{itemize}

\paragraph{Project Execution}\mbox{}\\[0.6em]
\textit{At project start}
\begin{itemize}
    \item[$\square$] Lead RSE: Update project planning in Ganttic
    \item[$\square$] Lead RSE: Organize regular project meetings
    \item[$\square$] Lead RSE: Organize internal coordination meetings
\end{itemize}
\textit{Throughout the project}
\begin{itemize}
    \item[$\square$] Lead RSE: Maintain project log
    \item[$\square$] Lead RSE: Monitor project hours and budget
    \item[$\square$] Lead RSE: Keep Ganttic planning up to date
    \item[$\square$] Lead RSE: Review and document risks and issues
    \item[$\square$] Lead RSE: Register project outputs, update RSD page
\end{itemize}
\textit{Milestones}
\begin{itemize}
    \item[$\square$] LA \& Lead RSE: Submit DMP within 3 months of project start
    \item[$\square$] SH: Organize annual project review
\end{itemize}

\paragraph{Project Closing}

\begin{itemize}
    \item[$\square$] LA: Submit end report
    \item[$\square$] Lead RSE \& SH: Complete scientific review
    \item[$\square$] Lead RSE: Update RSD page with final project information
    \item[$\square$] Finance: Complete financial review
    \item[$\square$] Finance: Archive final deliverables and documentation
    \item[$\square$] Finance: Send closing letter and formally close project
\end{itemize}
